/*
 * rk3588_gpio.c - RK3588 开发板 GPIO 控制示例（libgpiod v1.x）
 *
 * 适用：Ubuntu / Debian / Buildroot 等 Linux 系统，libgpiod 1.x
 *       （Rockchip 官方 Ubuntu 22.04 镜像自带该版本）
 *
 * 编译：make
 * 用法（大多数情况下需要 sudo，或把用户加入 gpio 组）：
 *   sudo ./rk3588_gpio GPIO4_A2 out 1      # 输出高电平
 *   sudo ./rk3588_gpio 4_A2 out 0          # 输出低电平
 *   sudo ./rk3588_gpio 4_A2 in             # 读取输入电平
 *   sudo ./rk3588_gpio 4_A2 blink 500      # 每 500ms 翻转，Ctrl+C 退出
 *
 * 说明：
 *   - 参数写成 "GPIO4_A2"、"4_A2"、"4a2" 都可以。
 *   - 默认 bank N 对应 /dev/gpiochipN；若板卡 DTS 改了 gpiochip 顺序，
 *     用环境变量覆盖，例如：RK_GPIO_CHIP=gpiochip4 ./rk3588_gpio ...
 *   - 具体哪个 GPIO 引到了排针，请查对应开发板原理图 / 引脚表
 *     （Rock 5B、Orange Pi 5、Radxa ROCK 5A 等的排针定义并不相同）。
 *   - 本例使用 libgpiod 1.x API；libgpiod 2.x（如部分 Ubuntu 24.04 镜像）
 *     需把 gpiod_chip_open_by_name / gpiod_line_request_output 等
 *     换成 gpiod_chip_open / gpiod_line_request 等新接口。
 */
#include <gpiod.h>
#include <errno.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

#include "rk3588_common.h"

static volatile sig_atomic_t keep_running = 1;

static void on_sigint(int signo)
{
	(void)signo;
	keep_running = 0;
}

static void usage(const char *prog)
{
	fprintf(stderr,
		"Usage:\n"
		"  %s <GPIO4_A2|4_A2|4a2> out [0|1]\n"
		"  %s <GPIO4_A2|4_A2|4a2> in\n"
		"  %s <GPIO4_A2|4_A2|4a2> blink [period_ms]\n",
		prog, prog, prog);
}

static void print_pin(unsigned int bank, unsigned int offset)
{
	printf("GPIO%u_%c%u", bank, 'A' + offset / 8u, offset % 8u);
}

int main(int argc, char **argv)
{
	unsigned int bank = 0, offset = 0;
	const char *chip_override;
	char chip_name[32];
	struct gpiod_chip *chip;
	struct gpiod_line *line;
	const char *mode;
	int ret = 1;
	int initial = 0, want = 0, period_ms = 500;

	if (argc < 3) {
		usage(argv[0]);
		return 1;
	}

	if (rk3588_parse_gpio(argv[1], &bank, &offset) != 0) {
		fprintf(stderr, "Invalid GPIO spec: %s\n", argv[1]);
		usage(argv[0]);
		return 1;
	}
	mode = argv[2];

	if (!strcmp(mode, "out")) {
		if (argc >= 4) {
			if (!strcmp(argv[3], "0"))
				initial = 0;
			else if (!strcmp(argv[3], "1"))
				initial = 1;
			else {
				fprintf(stderr, "value must be 0 or 1\n");
				return 1;
			}
			want = initial;
		}
	} else if (!strcmp(mode, "blink")) {
		if (argc >= 4) {
			period_ms = atoi(argv[3]);
			if (period_ms <= 0) {
				fprintf(stderr, "invalid period\n");
				return 1;
			}
		}
		signal(SIGINT, on_sigint);
	} else if (!strcmp(mode, "in")) {
		if (argc != 3) {
			usage(argv[0]);
			return 1;
		}
	} else {
		usage(argv[0]);
		return 1;
	}

	chip_override = getenv("RK_GPIO_CHIP");
	if (chip_override && *chip_override)
		snprintf(chip_name, sizeof(chip_name), "%s", chip_override);
	else
		snprintf(chip_name, sizeof(chip_name), "gpiochip%u", bank);

	chip = gpiod_chip_open_by_name(chip_name);
	if (!chip) {
		fprintf(stderr, "open %s failed: %s\n", chip_name, strerror(errno));
		return 1;
	}

	line = gpiod_chip_get_line(chip, offset);
	if (!line) {
		fprintf(stderr, "get line %u from %s failed: %s\n",
			offset, chip_name, strerror(errno));
		goto out_chip;
	}

	if (!strcmp(mode, "in")) {
		if (gpiod_line_request_input(line, "rk3588-gpio-demo") < 0) {
			fprintf(stderr, "request input failed: %s\n", strerror(errno));
			goto out_chip;
		}
		ret = gpiod_line_get_value(line);
		if (ret < 0) {
			fprintf(stderr, "read failed: %s\n", strerror(errno));
			ret = 1;
		} else {
			print_pin(bank, offset);
			printf(" = %d\n", ret);
			ret = 0;
		}
	} else {
		if (gpiod_line_request_output(line, "rk3588-gpio-demo", initial) < 0) {
			fprintf(stderr, "request output failed: %s\n", strerror(errno));
			goto out_chip;
		}

		if (!strcmp(mode, "out")) {
			if (gpiod_line_set_value(line, want) < 0) {
				fprintf(stderr, "set value failed: %s\n", strerror(errno));
				goto out_line;
			}
			print_pin(bank, offset);
			printf(" = %d\n", want);
			ret = 0;
		} else {
			int val = initial;

			printf("blinking, period=%dms, press Ctrl+C to stop\n", period_ms);
			while (keep_running) {
				val ^= 1;
				if (gpiod_line_set_value(line, val) < 0) {
					fprintf(stderr, "set value failed: %s\n", strerror(errno));
					goto out_line;
				}
				usleep((useconds_t)period_ms * 1000);
			}
			gpiod_line_set_value(line, 0);
			ret = 0;
		}
	}

out_line:
	gpiod_line_release(line);
out_chip:
	gpiod_chip_close(chip);
	return ret;
}
